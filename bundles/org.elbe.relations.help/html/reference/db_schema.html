Database schema

This page shows the SQL commands that create the tables needed to store the information created by the Relations application in a Derby database. The SQL commands for other databases (e.g. MySQL) are analogous.

Derby Database


CREATE TABLE tblTerm (
  TermID       BIGINT generated always as identity,
  sTitle       VARCHAR(99) not null,
  sText        CLOB,
  dtCreation   TIMESTAMP not null,
  dtMutation   TIMESTAMP not null,
  PRIMARY KEY (TermID)
);


CREATE INDEX idxTerm_01 ON tblTerm(sTitle);



CREATE TABLE tblText (
  TextID       BIGINT generated always as identity,
  sTitle       VARCHAR(150) not null,
  sText        CLOB,
  sAuthor      VARCHAR(100),
  sCoAuthors   VARCHAR(150),
  sSubtitle    VARCHAR(200),
  sYear        VARCHAR(15),
  sPublication VARCHAR(200),
  sPages       VARCHAR(20),
  nVolume      INT,
  nNumber      INT,
  sPublisher   VARCHAR(99),
  sPlace       VARCHAR(99),
  nType        INT,
  dtCreation   TIMESTAMP not null,
  dtMutation   TIMESTAMP not null,
  PRIMARY KEY (TextID)
);


CREATE INDEX idxText_01 ON tblText(sTitle);

CREATE INDEX idxText_02 ON tblText(sAuthor, sCoAuthors);


CREATE TABLE tblPerson (
  PersonID    BIGINT generated always as identity,
  sName       VARCHAR(99) not null,
  sFirstname  VARCHAR(50),
  sText       CLOB,
  sFrom       VARCHAR(30),
  sTo         VARCHAR(30),
  dtCreation  TIMESTAMP not null,
  dtMutation  TIMESTAMP not null,
  PRIMARY KEY (PersonID)
);


CREATE INDEX idxPerson_01 ON tblPerson(sName, sFirstname);

CREATE INDEX idxPerson_02 ON tblPerson(sFrom, sTo);

CREATE TABLE tblRelation (
  RelationID  BIGINT generated always as identity,
  nType1      SMALLINT not null,
  nItem1      BIGINT not null,
  nType2      SMALLINT not null,
  nItem2      BIGINT not null,
  PRIMARY KEY (RelationID)
);


CREATE INDEX idxRelation_01 ON tblRelation(nType1, nItem1);

CREATE INDEX idxRelation_02 ON tblRelation(nType2, nItem2);


MariaDB


CREATE TABLE `tblTerm` (
  `TermID` bigint(20) NOT NULL AUTO_INCREMENT,
  `sTitle` varchar(99) NOT NULL,
  `sText` text DEFAULT NULL,
  `dtCreation` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `dtMutation` timestamp NOT NULL DEFAULT '0000-00-00 00:00:00',
  PRIMARY KEY (`TermID`),
  KEY `idxTerm_01` (`sTitle`),
  KEY `idxTerm_02` (`dtCreation`),
  KEY `idxTerm_03` (`dtMutation`)
) ENGINE=InnoDB AUTO_INCREMENT=1627 DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;


CREATE TABLE `tblText` (
  `TextID` bigint(20) NOT NULL AUTO_INCREMENT,
  `sTitle` varchar(150) NOT NULL,
  `sText` text DEFAULT NULL,
  `sAuthor` varchar(100) DEFAULT NULL,
  `sCoAuthors` varchar(150) DEFAULT NULL,
  `sSubtitle` varchar(200) DEFAULT NULL,
  `sYear` varchar(15) DEFAULT NULL,
  `sPublication` varchar(200) DEFAULT NULL,
  `sPages` varchar(20) DEFAULT NULL,
  `nVolume` int(11) DEFAULT NULL,
  `nNumber` int(11) DEFAULT NULL,
  `sPublisher` varchar(99) DEFAULT NULL,
  `sPlace` varchar(99) DEFAULT NULL,
  `nType` int(11) DEFAULT NULL,
  `dtCreation` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `dtMutation` timestamp NOT NULL DEFAULT '0000-00-00 00:00:00',
  PRIMARY KEY (`TextID`),
  KEY `idxText_01` (`sTitle`),
  KEY `idxText_02` (`sAuthor`,`sCoAuthors`),
  KEY `idxText_03` (`dtCreation`),
  KEY `idxText_04` (`dtMutation`)
) ENGINE=InnoDB AUTO_INCREMENT=1837 DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;


CREATE TABLE `tblPerson` (
  `PersonID` bigint(20) NOT NULL AUTO_INCREMENT,
  `sName` varchar(99) NOT NULL,
  `sFirstname` varchar(50) DEFAULT NULL,
  `sText` text DEFAULT NULL,
  `sFrom` varchar(30) DEFAULT NULL,
  `sTo` varchar(30) DEFAULT NULL,
  `dtCreation` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `dtMutation` timestamp NOT NULL DEFAULT '0000-00-00 00:00:00',
  PRIMARY KEY (`PersonID`),
  KEY `idxPerson_01` (`sName`,`sFirstname`),
  KEY `idxPerson_02` (`sFrom`,`sTo`),
  KEY `idxPerson_03` (`dtCreation`),
  KEY `idxPerson_04` (`dtMutation`)
) ENGINE=InnoDB AUTO_INCREMENT=73 DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;


CREATE TABLE `tblRelation` (
  `RelationID` bigint(20) NOT NULL AUTO_INCREMENT,
  `nType1` smallint(6) NOT NULL,
  `nItem1` bigint(20) NOT NULL,
  `nType2` smallint(6) NOT NULL,
  `nItem2` bigint(20) NOT NULL,
  PRIMARY KEY (`RelationID`),
  KEY `idxRelation_01` (`nType1`,`nItem1`),
  KEY `idxRelation_02` (`nType2`,`nItem2`)
) ENGINE=InnoDB AUTO_INCREMENT=10059 DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;
